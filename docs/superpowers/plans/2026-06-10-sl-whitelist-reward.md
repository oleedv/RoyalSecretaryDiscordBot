# Squad Leader Whitelist Reward Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Reward consistent, high-quality squad leaders on the Royal Battalion Main server with auto-granted 7-day free whitelist via the existing `Squadleader whitelist` clan, plus a live Discord leaderboard and in-game progress notifications.

**Architecture:** Two-repo change. SquadJS gains an `sl-tracking` plugin that polls `ListSquads` every 30 s, persists per-squad tick snapshots to a new table, and aggregates a per-player-per-round summary at `ROUND_ENDED`. The Discord bot reads those aggregates on two crons: a 30-minute grant cron that writes/extends `WhitelistEntry` rows on the webpage DB, and a 3-hour leaderboard cron that maintains a live embed in a configured channel. The SquadJS plugin also fires an in-game `AdminWarn` when a squad first hits ≥5 members, gated on the player's webpage-DB whitelist status. A small DM SteamID register button on the bot is independent and ships any time after the tracking foundation.

**Tech Stack:** SquadJS (Node ES Modules, Sequelize, MariaDB, RCON), Discord bot (Bun, discord.js v14, Pino, mysql2 pools, Bun test), webpage DB (MariaDB, Prisma schema — read/written from bot pool only).

**Spec reference:** `docs/superpowers/specs/2026-06-10-sl-whitelist-reward-design.md`

```javascript
export function mapSquadsToTickRows({ serverId, matchId, tickAt, squads, playerCache }) {
  const rows = []
  for (const s of squads) {
    const leaderEos = s.leader?.eosID ?? null
    const leaderPlayerId = leaderEos ? (playerCache.get(leaderEos) ?? null) : null
    rows.push({
      server_id: serverId,
      match_id: matchId,
      tick_at: tickAt,
      team_id: s.teamID != null ? Number(s.teamID) : null,
      squad_id: s.squadID != null ? Number(s.squadID) : null,
      squad_name: s.squadName ?? null,
      leader_player_id: leaderPlayerId,
      member_count: Number(s.size ?? 0)
    })
  }
  return rows
}
```

```javascript
export function aggregateRound({ ticks, durationS }) {
  if (durationS == null || durationS < 1200) return []

  const halfRoundS = durationS / 2

  // Group by squad_id for Y, and by (player_id, squad_id) for player×squad contributions.
  const squadStrengthS = new Map()   // squad_id -> seconds where member_count >= 5
  const perPlayerSquad = new Map()   // player_id -> Map<squad_id, sl_seconds>

  for (const t of ticks) {
    if (t.squad_id == null) continue
    if ((t.member_count ?? 0) >= 5) {
      squadStrengthS.set(t.squad_id, (squadStrengthS.get(t.squad_id) ?? 0) + TICK_INTERVAL_S)
    }
    if (t.leader_player_id != null) {
      let bySquad = perPlayerSquad.get(t.leader_player_id)
      if (!bySquad) { bySquad = new Map(); perPlayerSquad.set(t.leader_player_id, bySquad) }
      bySquad.set(t.squad_id, (bySquad.get(t.squad_id) ?? 0) + TICK_INTERVAL_S)
    }
  }

  const rows = []
  for (const [player_id, bySquad] of perPlayerSquad) {
    let sl_tenure_s = 0
    let max_squad_strength_s = 0
    let qualifying_sl_s = 0
    let any_y = false
    for (const [squad_id, sl_s] of bySquad) {
      sl_tenure_s += sl_s
      const strength = squadStrengthS.get(squad_id) ?? 0
      if (strength > max_squad_strength_s) max_squad_strength_s = strength
      if (strength >= halfRoundS) {
        any_y = true
        // Within this squad, only ticks where it was strong AND player was SL count.
        // That's already enforced because we only added to bySquad if leader_player_id was this player
        // — but we need to limit to ticks where member_count >= 5. Recompute precisely:
        // Easier: count strong ticks in (player, squad) intersection.
        // Optimization deferred; compute below.
      }
    }

    // Re-walk ticks to compute qualifying_sl_s precisely (ticks where SL && strong && squad-Y-passed)
    for (const t of ticks) {
      if (t.leader_player_id !== player_id) continue
      if (t.squad_id == null) continue
      if ((t.member_count ?? 0) < 5) continue
      if ((squadStrengthS.get(t.squad_id) ?? 0) < halfRoundS) continue
      qualifying_sl_s += TICK_INTERVAL_S
    }

    const passed_y = any_y
    const passed_z = sl_tenure_s >= halfRoundS
    if (!passed_z) qualifying_sl_s = 0

    rows.push({
      player_id,
      sl_tenure_s,
      max_squad_strength_s,
      qualifying_sl_s,
      passed_y,
      passed_z
    })
  }
  return rows
}
```

```javascript
  async pollTick() {
    if (!this.server?.rcon) return
    let squads
    try {
      squads = await this.server.rcon.getSquads()
    } catch (e) {
      this.verbose(1, `pollTick: ListSquads failed: ${e?.message || e}`)
      return
    }
    if (!Array.isArray(squads) || squads.length === 0) return

    // Resolve unknown eos_ids to player_ids via squadjs_players (batch one query)
    const unknownEos = []
    for (const s of squads) {
      const eos = s.leader?.eosID
      if (eos && !this.playerCacheByEos.has(eos)) unknownEos.push(eos)
    }
    if (unknownEos.length > 0) {
      try {
        const rows = await this.options.database.query(
          'SELECT id, eos_id FROM squadjs_players WHERE eos_id IN (:eosIds)',
          { type: QueryTypes.SELECT, replacements: { eosIds: unknownEos } }
        )
        for (const r of rows) this.playerCacheByEos.set(r.eos_id, r.id)
      } catch (e) {
        this.verbose(1, `pollTick: player resolve failed: ${e?.message || e}`)
      }
    }

    const matchId = this.server.match?.id ?? null
    const serverId = this.server.id
    const tickAt = new Date()
    const playerCache = this.playerCacheByEos

    const rows = mapSquadsToTickRows({ serverId, matchId, tickAt, squads, playerCache })
    if (rows.length === 0) return

    try {
      await this.options.database.models.squadjs_squad_ticks.bulkCreate(rows)
    } catch (e) {
      this.verbose(1, `pollTick: bulkCreate failed: ${e?.message || e}`)
      return
    }

    this.verbose(2, `pollTick: inserted ${rows.length} tick rows`)

    // Detect <5 → >=5 transitions for warn dispatch (Task 8 wires the warn)
    for (const row of rows) {
      if (row.squad_id == null) continue
```

```javascript
import { SL_CLAN_ID } from './constants.js'

/**
 * Given a player's active WhitelistEntry rows (server='main'), classify them.
 * Returns { sl_entry, other_entry } where each is the first matching row or null.
 */
export function classifyWhitelist(entries) {
  let sl_entry = null
  let other_entry = null
  for (const e of entries) {
    if (e.clanId === SL_CLAN_ID && !sl_entry) sl_entry = e
    if (e.clanId !== SL_CLAN_ID && !other_entry) other_entry = e
  }
  return { sl_entry, other_entry }
}

/**
 * Decide what action to take given rolling hours and current entries.
 * Returns { action: 'grant' | 'extend' | 'skip', reason: string, sl_entry?: row }
 */
export function decideRewardAction({ rollingHours, entries, nowMs = Date.now(), extendThresholdHours = 24 }) {
  if (rollingHours < 5.0) return { action: 'skip', reason: 'under_threshold' }
  const { sl_entry, other_entry } = classifyWhitelist(entries)
  if (other_entry) return { action: 'skip', reason: 'has_other_whitelist' }
  if (!sl_entry) return { action: 'grant', reason: 'first_grant' }
  const remainingMs = new Date(sl_entry.expiresAt).getTime() - nowMs
  if (remainingMs < extendThresholdHours * 3600 * 1000) {
    return { action: 'extend', reason: 'near_expiry', sl_entry }
  }
  return { action: 'skip', reason: 'still_active' }
}
```

```javascript
  const candidates = await query(
    `SELECT player_id, steam_id, SUM(qualifying_sl_s) AS s
     FROM squadjs_sl_round_stats

    if (decision.action === 'grant') {
      const user = await query(
        `SELECT id, discordId FROM User WHERE steamId = ?`,
        [c.steam_id],
        'website'
      )
      const userId = user?.[0]?.id ?? null
      const newId = createId()
      if (DRY_RUN) {
        log.info({ steam_id: c.steam_id, rollingHours, action: 'grant' }, 'DRY: would grant')
      } else {
        await query(
          `INSERT INTO WhitelistEntry (id, steamId, server, clanId, userId, addedBy, reason, expiresAt, createdAt)
           VALUES (?, ?, ?, ?, ?, ?, ?, NOW() + INTERVAL ? DAY, NOW())`,
          [newId, c.steam_id, SL_SERVER, SL_CLAN_ID, userId, 'sl-reward-system',
           'Earned via Squad Leader rewards (5h+ rolling 7d)', SL_REWARD_DAYS],
          'website'
        )
        log.info({ steam_id: c.steam_id, rollingHours, entryId: newId }, 'granted SL whitelist')
        if (user?.[0]?.discordId) {
          try {
            const u = await client.users.fetch(user[0].discordId)
            await u.send(
              `Thanks for squadleading! You've earned 7 days of free whitelist on the Main server. ` +
              `Enjoy, and keep leading.`
            )
          } catch (e) {
            log.warn({ err: e?.message, discordId: user[0].discordId }, 'grant DM failed')
          }
        }
      }
      grants++
      continue
    }

    if (decision.action === 'extend') {
      if (DRY_RUN) {
        log.info({ steam_id: c.steam_id, entryId: decision.sl_entry.id, action: 'extend' }, 'DRY: would extend')
      } else {
        await query(
          `UPDATE WhitelistEntry SET expiresAt = expiresAt + INTERVAL ? DAY WHERE id = ?`,
          [SL_REWARD_DAYS, decision.sl_entry.id],
          'website'
        )
        log.info({ steam_id: c.steam_id, entryId: decision.sl_entry.id }, 'extended SL whitelist')
      }
      extensions++
    }
  }

  log.info({ candidates: candidates.length, grants, extensions, skips, dry: DRY_RUN }, 'sl-reward cron complete')
}

export const slRewardScheduler = scheduler('sl-reward-grant', INTERVAL_MS, tick)
```

```javascript
import { EmbedBuilder } from 'discord.js'
import { SL_CLAN_ID } from './constants.js'

/**
 * Pure: given top rows from SquadJS query, user records from webpage, and active whitelist entries,
 * produce display rows ready for the embed.
 */
export function buildLeaderboardRows({ topRows, users, whitelist }) {
  const userBySteam = new Map(users.map(u => [u.steamId, u]))
  const wlBySteam = new Map()
  for (const w of whitelist) {
    if (!wlBySteam.has(w.steamId)) wlBySteam.set(w.steamId, [])
    wlBySteam.get(w.steamId).push(w)
  }
  return topRows.map((r, i) => {
    const entries = wlBySteam.get(r.steam_id) ?? []
    let badge = '—'
    if (entries.some(e => e.clanId === SL_CLAN_ID)) badge = '✓ SL'
    else if (entries.length > 0) badge = '✓'
    const user = userBySteam.get(r.steam_id)
    return {
      rank: i + 1,
      displayName: user?.discordName ?? r.in_game_name,
      hours: Number(r.hours).toFixed(1),
      badge
    }
  })
}

/**
 * Pure: format display rows as a fixed-width code-block table for the embed body.
 */
export function formatLeaderboardTable(rows) {
  if (rows.length === 0) return 'No qualifying SL time in the last 7 days.'
  const header = `# Name${' '.repeat(28)}   Hrs    WL`
  const lines = rows.map(r => {
    const name = r.displayName.length > 28 ? r.displayName.slice(0, 27) + '…' : r.displayName
    const namePadded = name.padEnd(28, ' ')
    const rankPadded = String(r.rank).padStart(2, ' ')
    return `${rankPadded} ${namePadded}  ${r.hours.padStart(5, ' ')}  ${r.badge}`
  })
  return '```\n' + header + '\n' + lines.join('\n') + '\n```'
}

/**
 * Build an EmbedBuilder ready for send/edit.
 */
export function buildLeaderboardEmbed({ rows, nextUpdateUnix }) {
  return new EmbedBuilder()
    .setTitle('Squad Leader Rankings (rolling 7 days)')
    .setDescription('Top 20 SLs by qualifying SL time. SL ≥5h → 7 days free whitelist.')
    .addFields({ name: '​', value: formatLeaderboardTable(rows) })
    .setFooter({ text: 'Updated every 3 hours' })
    .setTimestamp(new Date(nextUpdateUnix * 1000))
    .setColor(0x2ecc71)
}
```

## Task 16: Implement the leaderboard cron (query, merge, post/edit)

```javascript
import { query } from '../../database/connection.js'
import logger from '../../logger.js'
import scheduler from '../../utils/scheduler.js'
import { getBotState, setBotState } from '../botState.js'
import { SL_SERVER } from './constants.js'
import { buildLeaderboardRows, buildLeaderboardEmbed } from './leaderboard.js'

const log = logger.child({ module: 'sl-reward-leaderboard' })

const STATE_KEY = 'sl_leaderboard_message'
const CHANNEL_ID = process.env.SL_LEADERBOARD_CHANNEL_ID
const INTERVAL_MS = Number(process.env.SL_LEADERBOARD_CRON_INTERVAL_MS) || 3 * 3600 * 1000

async function tick(client) {
  if (!CHANNEL_ID) { log.warn('SL_LEADERBOARD_CHANNEL_ID not set; skipping'); return }

  const topRows = await query(
    `SELECT p.id AS player_id, p.steam_id, p.name AS in_game_name,
            SUM(s.qualifying_sl_s)/3600.0 AS hours
     FROM squadjs_players p
     JOIN squadjs_sl_round_stats s ON s.player_id = p.id
     WHERE s.created_at > NOW() - INTERVAL 7 DAY
     GROUP BY p.id
     HAVING hours > 0
     ORDER BY hours DESC
     LIMIT 20`,
    [],
    'squadjs'
  )

  const steamIds = topRows.map(r => r.steam_id).filter(Boolean)
  let users = []
  let whitelist = []
  if (steamIds.length > 0) {
    const placeholders = steamIds.map(() => '?').join(',')
    users = await query(
      `SELECT steamId, discordName FROM User WHERE steamId IN (${placeholders})`,
      steamIds,
      'website'
    )
    whitelist = await query(
      `SELECT steamId, clanId FROM WhitelistEntry
       WHERE steamId IN (${placeholders}) AND server = ?
         AND (expiresAt IS NULL OR expiresAt > NOW())`,
      [...steamIds, SL_SERVER],
      'website'
    )
  }

  const rows = buildLeaderboardRows({ topRows, users, whitelist })
  const nextUpdateUnix = Math.floor((Date.now() + INTERVAL_MS) / 1000)
  const embed = buildLeaderboardEmbed({ rows, nextUpdateUnix })

  const channel = await client.channels.fetch(CHANNEL_ID).catch((e) => {
    log.error({ err: e?.message, CHANNEL_ID }, 'failed to fetch leaderboard channel')
    return null
  })
  if (!channel) return

  const state = await getBotState(STATE_KEY)
  if (state?.message_id) {
    try {
      const msg = await channel.messages.fetch(state.message_id)
      await msg.edit({ embeds: [embed] })
      log.info({ msg_id: state.message_id, rows: rows.length }, 'leaderboard edited')
      return
    } catch (e) {
      log.warn({ err: e?.message, msg_id: state.message_id }, 'stale message id; falling back to new post')
    }
  }

  const newMsg = await channel.send({ embeds: [embed] })
  await setBotState(STATE_KEY, { channel_id: channel.id, message_id: newMsg.id })
  log.info({ msg_id: newMsg.id, rows: rows.length }, 'leaderboard posted (new message)')
}

export const slLeaderboardScheduler = scheduler('sl-reward-leaderboard', INTERVAL_MS, tick)
```
